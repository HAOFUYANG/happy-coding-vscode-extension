### 核心文件详解

#### 扩展入口 (extension.ts)
```typescript
import "reflect-metadata";
import * as vscode from "vscode";
import { ContextService } from "@/service/context.service";
import {
  registerControllers,
  registerServices,
} from "cec-client-server/decorator";

// 导入所有控制器
import { GitController } from "./controller/git.controller";
import { AxiosController } from "./controller/axios.controller";
import { UserController } from "./controller/user.controller";
import { CodingController } from "./controller/coding.controller";
import { CliController } from "./controller/cli.controller";
import { PlaywrightController } from "./controller/script.controller";

export function activate(context: vscode.ExtensionContext): void {
  // 1. 注册全局上下文
  ContextService.register(context);
  
  // 2. 注册控制器 (RPC 调用接口)
  registerControllers([
    GitController,
    AxiosController,
    UserController,
    CodingController,
    CliController,
    PlaywrightController,
  ]);
  
  // 3. 注册服务 (事件订阅接口)
  registerServices([CodingController]);
}
```

  protected context: vscode.ExtensionContext;

    this.context = ContextService.getContext();
  }
  
  protected abstract initialize(): Promise<void>;
  
  protected handleError(error: Error, operation: string): void {

export class ContextService {
  private static context: vscode.ExtensionContext;
  
  static register(context: vscode.ExtensionContext): void {
    this.context = context;
  }

    if (!this.context) {

    return this.context.extensionPath;

    return this.context.globalState;
  }
  
  static getWorkspaceState(): vscode.Memento {
    return this.context.workspaceState;
  }
}

##### HTTP 服务 (axios.service.ts)
```typescript
import axios, { AxiosInstance, AxiosRequestConfig } from 'axios';

export class AxiosService {
  private instance: AxiosInstance;
  
  constructor() {
    this.instance = axios.create({
      timeout: 10000,
      headers: {
        'Content-Type': 'application/json',
      },
    });
    
    this.setupInterceptors();
  }
  
  private setupInterceptors(): void {
    // 请求拦截器
    this.instance.interceptors.request.use(
      (config) => {
        console.log(`[HTTP] ${config.method?.toUpperCase()} ${config.url}`);
        return config;
      },
      (error) => Promise.reject(error)
    );
    
    // 响应拦截器
    this.instance.interceptors.response.use(
      (response) => response,
      (error) => {
        console.error('[HTTP] Request failed:', error.message);
        return Promise.reject(error);
      }
    );
  }
  
  async get<T>(url: string, config?: AxiosRequestConfig): Promise<T> {
    const response = await this.instance.get(url, config);
    return response.data;
  }
  
  async post<T>(url: string, data?: any, config?: AxiosRequestConfig): Promise<T> {
    const response = await this.instance.post(url, data, config);
    return response.data;
  }
}
```

##### CLI 工具函数 (utils/happyCliUtils.ts)
```typescript
import { execSync } from 'child_process';
import * as semver from 'semver';

export interface NodeVersionResult {
  installed: boolean;
  version?: string;
  meetRequirement: boolean;
}

export interface CliInstallResult {
  installed: boolean;
  version?: string;
  globalPath?: string;
}

export function checkNodeVersion(minVersion = '14.0.0'): NodeVersionResult {
  try {
    const version = process.version;
    const meetRequirement = semver.gte(version, minVersion);
    
    return {
      installed: true,
      version,
      meetRequirement
    };
  } catch (error) {
    return {
      installed: false,
      meetRequirement: false
    };
  }
}

export function checkHappyCliInstalled(): CliInstallResult {
  try {
    const result = execSync('npm list -g @happy.cli/cli --depth=0', {
      encoding: 'utf8',
      stdio: 'pipe'
    });
    
    const versionMatch = result.match(/@happy\.cli\/cli@(\d+\.\d+\.\d+)/);
    const version = versionMatch ? versionMatch[1] : undefined;
    
    return {
      installed: true,
      version,
      globalPath: extractGlobalPath(result)
    };
  } catch (error) {
    return {
      installed: false
    };
  }
}

function extractGlobalPath(npmOutput: string): string | undefined {
  const lines = npmOutput.split('\n');
  const globalLine = lines.find(line => line.includes('node_modules'));
  return globalLine?.trim();
}
```

#### Hook 架构模式
```typescript
// hooks/useAsyncOperation.ts
export const useAsyncOperation = <T>() => {
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [data, setData] = useState<T | null>(null);
  
  const execute = useCallback(async (operation: () => Promise<T>) => {
    setLoading(true);
    setError(null);
    
    try {
      const result = await operation();
      setData(result);
      return result;
    } catch (err) {
      const errorMessage = err instanceof Error ? err.message : '操作失败';
      setError(errorMessage);
      throw err;
    } finally {
      setLoading(false);
    }
  }, []);
  
  return { loading, error, data, execute };
};

// hooks/useCoding.ts
export const useCoding = () => {
  const [acceptDetails, setAcceptDetails] = useState([]);
  const { loading, execute } = useAsyncOperation();
  
  useEffect(() => {
    // 订阅事件
    const unsubscribeUpdates = subscribe('generationUpdates', setAcceptDetails);
    const unsubscribeStatus = subscribe('generationStatus', (data) => {
      // 处理状态更新
    });
    
    return () => {
      unsubscribeUpdates();
      unsubscribeStatus();
    };
  }, []);
  
  const startCoding = useCallback((params) => {
    return execute(() => call('Coding', 'start', params));
  }, [execute]);
  
  return {
    acceptDetails,
    loading,
    startCoding,
    // ... 其他方法
  };
};
```

## 文件命名规范

### 命名约定

#### TypeScript 文件
- **控制器**: `*.controller.ts`
- **服务**: `*.service.ts`
- **工具函数**: `*.utils.ts`
- **类型定义**: `*.types.ts`
- **常量**: `*.constants.ts`

#### React 组件
- **组件文件**: `ComponentName.tsx` (PascalCase)
- **Hook 文件**: `useHookName.ts` (camelCase)
- **工具文件**: `utilName.ts` (camelCase)

#### 测试文件
- **单元测试**: `*.test.ts`
- **集成测试**: `*.integration.test.ts`
- **端到端测试**: `*.e2e.test.ts`
- **Playwright 测试**: `*.spec.ts`

### 目录命名
- 使用 kebab-case: `my-component/`
- 功能分组: `components/`, `services/`, `utils/`
- 按模块分组: `git/`, `cli/`, `coding/`
