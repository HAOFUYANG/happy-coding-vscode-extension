```typescript
// utils/performance.ts
export function debounce<T extends (...args: any[]) => any>(
  func: T,
  wait: number,
  immediate = false
): (...args: Parameters<T>) => void {
  let timeout: NodeJS.Timeout | null = null;
  
  return function executedFunction(...args: Parameters<T>) {
    const later = () => {
      timeout = null;
      if (!immediate) func(...args);
    };
    
    const callNow = immediate && !timeout;
    
    if (timeout) clearTimeout(timeout);
    timeout = setTimeout(later, wait);
    
    if (callNow) func(...args);
  };
}

export function throttle<T extends (...args: any[]) => any>(
  func: T,
  limit: number
): (...args: Parameters<T>) => void {
  let inThrottle: boolean;
  
  return function executedFunction(...args: Parameters<T>) {
    if (!inThrottle) {
      func.apply(this, args);
      inThrottle = true;
      setTimeout(() => inThrottle = false, limit);
    }
  };
}

// 使用示例

  // 防抖的状态更新

    this.subscribers.forEach(cb => cb(data));
  }, 100);
  
  // 节流的进度更新
  private throttledProgressUpdate = throttle((progress: number) => {
    this.statusSubscribers.forEach(cb => cb({ progress }));
  }, 50);
}
```

```typescript
class EventPool {
  private pools = new Map<string, any[]>();
  private maxPoolSize = 100;
  
  get<T>(type: string, factory: () => T): T {
    const pool = this.pools.get(type) || [];
    
    if (pool.length > 0) {
      return pool.pop() as T;
    }
    
    return factory();
  }
  
  release<T>(type: string, obj: T, reset?: (obj: T) => void): void {
    const pool = this.pools.get(type) || [];
    
    if (pool.length < this.maxPoolSize) {
      if (reset) reset(obj);
      pool.push(obj);
      this.pools.set(type, pool);
    }
  }
  
  clear(type?: string): void {
    if (type) {
      this.pools.delete(type);
    } else {
      this.pools.clear();
    }
  }
}
```

```typescript
// utils/concurrency.ts
class ConcurrencyLimiter {
  private running = 0;
  private queue: Array<() => Promise<any>> = [];
  
  constructor(private limit: number) {}
  
  async execute<T>(task: () => Promise<T>): Promise<T> {
    return new Promise((resolve, reject) => {
      this.queue.push(async () => {
        try {
          const result = await task();
          resolve(result);
        } catch (error) {
          reject(error);
        }
      });
      
      this.processQueue();
    });
  }
  
  private async processQueue(): Promise<void> {
    if (this.running >= this.limit || this.queue.length === 0) {
      return;
    }
    
    this.running++;
    const task = this.queue.shift()!;
    
    try {
      await task();
    } finally {
      this.running--;
      this.processQueue();
    }
  }
}

// 使用示例
const gitOperationLimiter = new ConcurrencyLimiter(3);
const fileOperationLimiter = new ConcurrencyLimiter(5);

  async getRemotesWithPath(): Promise<{ remotes: string[]; cwd: string }> {
    return gitOperationLimiter.execute(async () => {
      // Git 操作实现
      return this.executeGitCommand('remote -v');
    });
```

```typescript
// utils/batch-processor.ts
class BatchProcessor<T, R> {
  private batch: T[] = [];
  private timer: NodeJS.Timeout | null = null;
  private resolvers: Array<(result: R[]) => void> = [];
  
  constructor(
    private processor: (items: T[]) => Promise<R[]>,
    private batchSize: number = 10,
    private delay: number = 100
  ) {}
  
  async add(item: T): Promise<R> {
    return new Promise<R>((resolve) => {
      this.batch.push(item);
      this.resolvers.push(resolve as any);
      
      if (this.batch.length >= this.batchSize) {
        this.flush();
      } else if (!this.timer) {
        this.timer = setTimeout(() => this.flush(), this.delay);
      }
    });
  }
  
  private async flush(): Promise<void> {
    if (this.batch.length === 0) return;
    
    const currentBatch = this.batch.splice(0);
    const currentResolvers = this.resolvers.splice(0);
    
    if (this.timer) {
      clearTimeout(this.timer);
      this.timer = null;
    }
    
    try {
      const results = await this.processor(currentBatch);
      currentResolvers.forEach((resolve, index) => {
        resolve(results[index]);
      });
    } catch (error) {
      currentResolvers.forEach((resolve) => {
        resolve(error as any);
      });
    }
  }
}

// 使用示例
const fileProcessor = new BatchProcessor(
  async (filePaths: string[]) => {
    // 批量处理文件操作
    return Promise.all(filePaths.map(path => fs.stat(path)));
  },
  20,  // 批处理大小
  50   // 延迟时间
);
```

```typescript
// utils/memory-manager.ts
class MemoryManager {
  private cleanupTasks = new Set<() => void>();
  private intervals = new Set<NodeJS.Timeout>();
  private timeouts = new Set<NodeJS.Timeout>();
  private subscriptions = new Set<() => void>();
  
  // 注册清理任务
  registerCleanup(cleanup: () => void): void {
    this.cleanupTasks.add(cleanup);
  }
  
  // 管理定时器
  setInterval(callback: () => void, ms: number): NodeJS.Timeout {
    const interval = setInterval(callback, ms);
    this.intervals.add(interval);
    return interval;
  }
  
  setTimeout(callback: () => void, ms: number): NodeJS.Timeout {
    const timeout = setTimeout(() => {
      callback();
      this.timeouts.delete(timeout);
    }, ms);
    this.timeouts.add(timeout);
    return timeout;
  }
  
  // 管理订阅
  addSubscription(unsubscribe: () => void): void {
    this.subscriptions.add(unsubscribe);
  }
  
  // 清理所有资源
  cleanup(): void {
    // 清理定时器
    this.intervals.forEach(interval => clearInterval(interval));
    this.timeouts.forEach(timeout => clearTimeout(timeout));
    
    // 清理订阅
    this.subscriptions.forEach(unsubscribe => unsubscribe());
    
    // 执行自定义清理任务
    this.cleanupTasks.forEach(cleanup => {
      try {
        cleanup();
      } catch (error) {
        console.error('Cleanup task failed:', error);
      }
    });
    
    // 清空集合
    this.intervals.clear();
    this.timeouts.clear();
    this.subscriptions.clear();
    this.cleanupTasks.clear();
  }
}

// 控制器基类集成内存管理
abstract class BaseController {
  protected memoryManager = new MemoryManager();
  
  dispose(): void {
    this.memoryManager.cleanup();
  }
}
```

```typescript
// utils/weak-cache.ts
class WeakCache<K extends object, V> {
  private cache = new WeakMap<K, V>();
  private keyRefs = new Set<WeakRef<K>>();
  private cleanupRegistry = new FinalizationRegistry((key: K) => {
    // 清理已被垃圾回收的键的引用
    this.keyRefs.forEach(ref => {
      if (ref.deref() === undefined) {
        this.keyRefs.delete(ref);
      }
    });
  });
  
  set(key: K, value: V): void {
    this.cache.set(key, value);
    const keyRef = new WeakRef(key);
    this.keyRefs.add(keyRef);
    this.cleanupRegistry.register(key, key);
  }
  
  get(key: K): V | undefined {
    return this.cache.get(key);
  }
  
  has(key: K): boolean {
    return this.cache.has(key);
  }
  
  delete(key: K): boolean {
    return this.cache.delete(key);
  }
  
  // 手动清理已失效的引用
  cleanup(): void {
    this.keyRefs.forEach(ref => {
      if (ref.deref() === undefined) {
        this.keyRefs.delete(ref);
      }
    });
  }
}
```

```typescript
// utils/circular-buffer.ts
class CircularBuffer<T> {
  private buffer: T[];
  private head = 0;
  private tail = 0;
  private size = 0;
  
  constructor(private capacity: number) {
    this.buffer = new Array(capacity);
  }
  
  push(item: T): void {
    this.buffer[this.tail] = item;
    this.tail = (this.tail + 1) % this.capacity;
    
    if (this.size < this.capacity) {
      this.size++;
    } else {
      this.head = (this.head + 1) % this.capacity;
    }
  }
  
  pop(): T | undefined {
    if (this.size === 0) return undefined;
    
    const item = this.buffer[this.head];
    this.head = (this.head + 1) % this.capacity;
    this.size--;
    
    return item;
  }
  
  peek(): T | undefined {
    return this.size > 0 ? this.buffer[this.head] : undefined;
  }
  
  toArray(): T[] {
    const result: T[] = [];
    for (let i = 0; i < this.size; i++) {
      const index = (this.head + i) % this.capacity;
      result.push(this.buffer[index]);
    }
    return result;
  }
  
  clear(): void {
    this.head = 0;
    this.tail = 0;
    this.size = 0;
  }
  
  get length(): number {
    return this.size;
  }
  
  get isFull(): boolean {
    return this.size === this.capacity;
  }
}

// 使用示例 - 日志缓冲区
class LogManager {
  private logBuffer = new CircularBuffer<LogEntry>(1000);
  
  addLog(level: string, message: string): void {
    this.logBuffer.push({
      level,
      message,
      timestamp: Date.now(),
    });
  }
  
  getRecentLogs(count: number = 100): LogEntry[] {
    const allLogs = this.logBuffer.toArray();
    return allLogs.slice(-count);
  }
}
```

```typescript
  items: T[];
  itemHeight: number;
  containerHeight: number;
  renderItem: (item: T, index: number) => React.ReactNode;
}

function VirtualList<T>({ 
  items, 
  itemHeight, 
  containerHeight, 
  renderItem 
}: VirtualListProps<T>) {
  const [scrollTop, setScrollTop] = useState(0);
  const containerRef = useRef<HTMLDivElement>(null);
  
  // 计算可见范围
  const visibleStart = Math.floor(scrollTop / itemHeight);
  const visibleEnd = Math.min(
    visibleStart + Math.ceil(containerHeight / itemHeight) + 1,
    items.length
  );
  
  // 可见项目
  const visibleItems = items.slice(visibleStart, visibleEnd);
  
  // 滚动处理
  const handleScroll = useCallback((e: React.UIEvent<HTMLDivElement>) => {
    setScrollTop(e.currentTarget.scrollTop);
  }, []);
  
  return (
    <div
      ref={containerRef}
      style={{ height: containerHeight, overflow: 'auto' }}
      onScroll={handleScroll}
    >
      {/* 总高度占位 */}
      <div style={{ height: items.length * itemHeight, position: 'relative' }}>
        {/* 可见项目 */}
        <div
          style={{
            transform: `translateY(${visibleStart * itemHeight}px)`,
            position: 'absolute',

            left: 0,
            right: 0,
```

```typescript
    loading: false,
    error: null,
  });

      loading: false,
      error: null,
    });

    ...slice,
    updateData,
    setLoading,
    setError,
    reset,
  };
```

```typescript
// utils/request-batcher.ts
class RequestBatcher {
  private batches = new Map<string, {
    requests: Array<{
      params: any;
      resolve: (result: any) => void;
      reject: (error: any) => void;
    }>;
    timer: NodeJS.Timeout;
  }>();
  
  constructor(private batchDelay: number = 50) {}
  
  async batch<T>(
    key: string,
    params: any,
    batchProcessor: (allParams: any[]) => Promise<T[]>
  ): Promise<T> {
    return new Promise((resolve, reject) => {
      let batch = this.batches.get(key);
      
      if (!batch) {
        batch = {
          requests: [],
          timer: setTimeout(() => this.processBatch(key, batchProcessor), this.batchDelay),
        };
        this.batches.set(key, batch);
      }
      
      batch.requests.push({ params, resolve, reject });
    });
  }
  
  private async processBatch<T>(
    key: string,
    batchProcessor: (allParams: any[]) => Promise<T[]>
  ): Promise<void> {
    const batch = this.batches.get(key);
    if (!batch) return;
    
    this.batches.delete(key);
    
    try {
      const allParams = batch.requests.map(req => req.params);
      const results = await batchProcessor(allParams);
      
      batch.requests.forEach((req, index) => {
        req.resolve(results[index]);
      });
    } catch (error) {
      batch.requests.forEach(req => {
        req.reject(error);
      });
    }
  }
}

// 使用示例
```

#### 缓存策略
```typescript
// utils/cache-manager.ts
interface CacheEntry<T> {
  data: T;
  timestamp: number;
  ttl: number;
}

class CacheManager {
  private cache = new Map<string, CacheEntry<any>>();
  private maxSize: number;
  private cleanupInterval: NodeJS.Timeout;
  
  constructor(maxSize: number = 1000, cleanupIntervalMs: number = 60000) {
    this.maxSize = maxSize;
    this.cleanupInterval = setInterval(() => this.cleanup(), cleanupIntervalMs);
  }
  
  set<T>(key: string, data: T, ttl: number = 300000): void {
    // 如果缓存已满，删除最旧的条目
    if (this.cache.size >= this.maxSize) {
      const oldestKey = this.cache.keys().next().value;
      this.cache.delete(oldestKey);
    }
    
    this.cache.set(key, {
      data,
      timestamp: Date.now(),
      ttl,
    });
  }
  
  get<T>(key: string): T | null {
    const entry = this.cache.get(key);
    if (!entry) return null;
    
    // 检查是否过期
    if (Date.now() - entry.timestamp > entry.ttl) {
      this.cache.delete(key);
      return null;
    }
    
    return entry.data;
  }
  
  has(key: string): boolean {
    const entry = this.cache.get(key);
    if (!entry) return false;
    
    // 检查是否过期
    if (Date.now() - entry.timestamp > entry.ttl) {
      this.cache.delete(key);
      return false;
    }
    
    return true;
  }
  
  delete(key: string): boolean {
    return this.cache.delete(key);
  }
  
  clear(): void {
    this.cache.clear();
  }
  
  private cleanup(): void {
    const now = Date.now();
    for (const [key, entry] of this.cache.entries()) {
      if (now - entry.timestamp > entry.ttl) {
        this.cache.delete(key);
      }
    }
  }
  
  dispose(): void {
    clearInterval(this.cleanupInterval);
    this.clear();
  }
}

// 全局缓存实例
const globalCache = new CacheManager();

// 带缓存的 API 调用
async function cachedApiCall<T>(
  key: string,
  apiCall: () => Promise<T>,
  ttl: number = 300000
): Promise<T> {
  // 尝试从缓存获取
  const cached = globalCache.get<T>(key);
  if (cached !== null) {
    return cached;
  }
  
  // 执行 API 调用
  const result = await apiCall();
  
  // 存入缓存
  globalCache.set(key, result, ttl);
  
  return result;
}
```

```typescript
// utils/performance-monitor.ts
interface PerformanceMetric {
  name: string;
  value: number;
  timestamp: number;
  tags?: Record<string, string>;
}

class PerformanceMonitor {
  private metrics: PerformanceMetric[] = [];
  private timers = new Map<string, number>();
  
  // 开始计时
  startTimer(name: string, tags?: Record<string, string>): void {
    const key = this.getTimerKey(name, tags);
    this.timers.set(key, performance.now());
  }
  
  // 结束计时并记录
  endTimer(name: string, tags?: Record<string, string>): number {
    const key = this.getTimerKey(name, tags);
    const startTime = this.timers.get(key);
    
    if (startTime === undefined) {
      console.warn(`Timer not found: ${key}`);
      return 0;
    }
    
    const duration = performance.now() - startTime;
    this.timers.delete(key);
    
    this.recordMetric(name, duration, tags);
    return duration;
  }
  
  // 记录指标
  recordMetric(name: string, value: number, tags?: Record<string, string>): void {
    this.metrics.push({
      name,
      value,
      timestamp: Date.now(),
      tags,
    });
    
    // 限制指标数量
    if (this.metrics.length > 10000) {
      this.metrics = this.metrics.slice(-5000);
    }
  }
  
  // 获取指标统计
  getStats(name: string, timeRange?: number): {
    count: number;
    avg: number;
    min: number;
    max: number;
    p95: number;
  } {
    const now = Date.now();
    const cutoff = timeRange ? now - timeRange : 0;
    
    const relevantMetrics = this.metrics.filter(
      metric => metric.name === name && metric.timestamp > cutoff
    );
    
    if (relevantMetrics.length === 0) {
      return { count: 0, avg: 0, min: 0, max: 0, p95: 0 };
    }
    
    const values = relevantMetrics.map(m => m.value).sort((a, b) => a - b);
    const count = values.length;
    const sum = values.reduce((a, b) => a + b, 0);
    const avg = sum / count;
    const min = values[0];
    const max = values[count - 1];
    const p95Index = Math.floor(count * 0.95);
    const p95 = values[p95Index] || max;
    
    return { count, avg, min, max, p95 };
  }
  
  private getTimerKey(name: string, tags?: Record<string, string>): string {
    if (!tags) return name;
    const tagString = Object.entries(tags)
      .sort(([a], [b]) => a.localeCompare(b))
      .map(([k, v]) => `${k}=${v}`)
      .join(',');
    return `${name}[${tagString}]`;
  }
  
  // 导出指标数据
  exportMetrics(): PerformanceMetric[] {
    return [...this.metrics];
  }
  
  // 清理旧指标
  cleanup(maxAge: number = 3600000): void {
    const cutoff = Date.now() - maxAge;
    this.metrics = this.metrics.filter(m => m.timestamp > cutoff);
  }
}

// 装饰器用法
function measurePerformance(name: string, tags?: Record<string, string>) {
  return function (target: any, propertyKey: string, descriptor: PropertyDescriptor) {

      }
    };
    
    return descriptor;
  };
}
```

```typescript
// utils/memory-monitor.ts
class MemoryMonitor {
  private samples: Array<{
    timestamp: number;
    usage: NodeJS.MemoryUsage;
  }> = [];
  
  private interval: NodeJS.Timeout;
  
  constructor(sampleInterval: number = 30000) {
    this.interval = setInterval(() => {
      this.takeSample();
    }, sampleInterval);
  }
  
  private takeSample(): void {
    const usage = process.memoryUsage();
    this.samples.push({
      timestamp: Date.now(),
      usage,
    });
    
    // 保留最近 24 小时的样本
    const cutoff = Date.now() - 24 * 60 * 60 * 1000;
    this.samples = this.samples.filter(s => s.timestamp > cutoff);
  }
  
  getCurrentUsage(): NodeJS.MemoryUsage {
    return process.memoryUsage();
  }
  
  getUsageStats(timeRange?: number): {
    current: NodeJS.MemoryUsage;
    peak: NodeJS.MemoryUsage;
    average: NodeJS.MemoryUsage;
  } {
    const now = Date.now();
    const cutoff = timeRange ? now - timeRange : 0;
    
    const relevantSamples = this.samples.filter(s => s.timestamp > cutoff);
    
    if (relevantSamples.length === 0) {
      const current = this.getCurrentUsage();
      return { current, peak: current, average: current };
    }
    
    const current = this.getCurrentUsage();
    
    // 计算峰值
    const peak = relevantSamples.reduce((max, sample) => ({
      rss: Math.max(max.rss, sample.usage.rss),
      heapTotal: Math.max(max.heapTotal, sample.usage.heapTotal),
      heapUsed: Math.max(max.heapUsed, sample.usage.heapUsed),
      external: Math.max(max.external, sample.usage.external),
      arrayBuffers: Math.max(max.arrayBuffers, sample.usage.arrayBuffers),
    }), relevantSamples[0].usage);
    
    // 计算平均值
    const sum = relevantSamples.reduce((acc, sample) => ({
      rss: acc.rss + sample.usage.rss,
      heapTotal: acc.heapTotal + sample.usage.heapTotal,
      heapUsed: acc.heapUsed + sample.usage.heapUsed,
      external: acc.external + sample.usage.external,
      arrayBuffers: acc.arrayBuffers + sample.usage.arrayBuffers,
    }), { rss: 0, heapTotal: 0, heapUsed: 0, external: 0, arrayBuffers: 0 });
    
    const count = relevantSamples.length;
    const average = {
      rss: Math.round(sum.rss / count),
      heapTotal: Math.round(sum.heapTotal / count),
      heapUsed: Math.round(sum.heapUsed / count),
      external: Math.round(sum.external / count),
      arrayBuffers: Math.round(sum.arrayBuffers / count),
    };
    
    return { current, peak, average };
  }
  
  // 检查内存泄漏
  checkMemoryLeak(threshold: number = 100 * 1024 * 1024): boolean {
    if (this.samples.length < 10) return false;
    
    const recent = this.samples.slice(-10);
    const trend = recent.map(s => s.usage.heapUsed);
    
    // 检查是否持续增长
    let increasing = 0;
    for (let i = 1; i < trend.length; i++) {
      if (trend[i] > trend[i - 1]) {
        increasing++;
      }
    }
    
    // 如果 80% 的样本都在增长，且增长超过阈值
    const isIncreasing = increasing / (trend.length - 1) > 0.8;
    const growth = trend[trend.length - 1] - trend[0];
    
    return isIncreasing && growth > threshold;
  }
  
  dispose(): void {
    clearInterval(this.interval);
  }
}

// 全局内存监控
const memoryMonitor = new MemoryMonitor();
```
